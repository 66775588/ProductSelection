```python
def claim_job(job_id: str, category: str, platform: str) -> bool:
    """原子抢占单任务锁。已有 running 任务则失败。"""
    init_db()
    with _connect() as conn:
        if conn.execute(
            "SELECT 1 FROM running_jobs WHERE status='running'"
        ).fetchone():
            return False
        try:
            conn.execute(
                "INSERT INTO running_jobs (job_id, category, platform, started_at, status) "
                "VALUES (?, ?, ?, ?, 'running')",
                (job_id, category, platform, datetime.now().isoformat()),
            )
            conn.commit()
            return True
        except sqlite3.IntegrityError:
            return False


def get_running_job() -> dict[str, Any] | None:
    init_db()
    with _connect() as conn:
        conn.row_factory = sqlite3.Row
        row = conn.execute(
            "SELECT job_id, category, platform, started_at FROM running_jobs WHERE status='running'"
        ).fetchone()
        return dict(row) if row else None


def release_job(job_id: str) -> None:
    init_db()
    with _connect() as conn:
        conn.execute("DELETE FROM running_jobs WHERE job_id=?", (job_id,))
        conn.execute("DELETE FROM job_progress WHERE job_id=?", (job_id,))
        conn.commit()


def stale_jobs(older_than_minutes: int = 30) -> list[str]:
    init_db()
    cutoff = (datetime.now() - timedelta(minutes=older_than_minutes)).isoformat()
    with _connect() as conn:
        rows = conn.execute(
            "SELECT job_id FROM running_jobs WHERE status='running' AND started_at < ?",
            (cutoff,),
        ).fetchall()
        return [r[0] for r in rows]


def append_progress(job_id: str, node_key: str) -> None:
    init_db()
    with _connect() as conn:
        conn.execute(
            "INSERT OR IGNORE INTO job_progress (job_id, node_key, completed_at) VALUES (?, ?, ?)",
            (job_id, node_key, datetime.now().isoformat()),
        )
        conn.commit()


def load_progress(job_id: str) -> list[str]:
    init_db()
    with _connect() as conn:
        rows = conn.execute(
            "SELECT node_key FROM job_progress WHERE job_id=? ORDER BY rowid", (job_id,)
        ).fetchall()
        return [r[0] for r in rows]
```

```python
# src/dianshang/server/paths.py
"""报告文件路径安全工具：锁定 data/reports 目录，防穿越。"""
from __future__ import annotations

from pathlib import Path

from dianshang.config import settings


def sanitize_report_path(filename: str) -> Path | None:
    """校验 filename 合法且 resolve 后仍在 reports_dir 内。合法返回绝对 Path。"""
    if not filename or not isinstance(filename, str):
        return None
    # 严禁穿越与绝对路径
    if "\\" in filename or "/" in filename or filename.startswith("."):
        return None
    reports_dir: Path = settings.paths.reports_dir.resolve()
    try:
        resolved = (reports_dir / filename).resolve()
    except (ValueError, OSError):
        return None
    if resolved.parent != reports_dir:
        return None
    return resolved
```

```python
# src/dianshang/server/config_store.py
"""运行时覆盖：校验 / 增量合并 / 文件锁持久化 / 原地应用到 settings 单例。"""
from __future__ import annotations

import json
from typing import Any

import portalocker

from dianshang.config import settings

OVERRIDES_PATH = settings.paths.data_dir / "runtime_overrides.json"

WEIGHT_KEYS = ["market_potential", "competition", "profit", "pain_points", "risk"]
PROFIT_RATE_KEYS = ["platform_commission_rate", "logistics_rate", "promotion_rate",
                    "refund_loss_rate", "tax_rate"]
ALLOWED_TOP = {"scoring_weights", "profit", "crawl"}


def validate_overrides(overrides: dict[str, Any]) -> list[str]:
    errs: list[str] = []
    if not isinstance(overrides, dict):
        return ["根必须是对象"]
    for key in overrides:
        if key not in ALLOWED_TOP:
            errs.append(f"未知字段: {key}")

    sw = overrides.get("scoring_weights")
    if sw is not None:
        if not isinstance(sw, dict) or set(sw.keys()) != set(WEIGHT_KEYS):
            errs.append("scoring_weights 必须包含且仅包含五个权重字段")
        elif abs(sum(sw.values()) - 1.0) > 0.01:
            errs.append(f"权重之和必须为 1.0（当前 {round(sum(sw.values()), 4)}）")

    pf = overrides.get("profit")
    if pf is not None:
        if not isinstance(pf, dict):
            errs.append("profit 必须是对象")
        else:
            if "profit_threshold" in pf and not (0 <= pf["profit_threshold"] <= 100):
                errs.append("profit_threshold 必须在 0–100")
            for k in PROFIT_RATE_KEYS:
                if k in pf and not (0 <= pf[k] <= 1):
                    errs.append(f"{k} 必须在 0–1")
            if "packaging_cost" in pf and pf["packaging_cost"] <= 0:
                errs.append("packaging_cost 必须 > 0")

    cr = overrides.get("crawl")
    if cr is not None and "force_mock" in cr and not isinstance(cr["force_mock"], bool):
        errs.append("force_mock 必须是布尔")
    return errs


def load_overrides() -> dict[str, Any]:
    if not OVERRIDES_PATH.exists():
        return {}
    try:
        with portalocker.Lock(str(OVERRIDES_PATH), "r", timeout=5) as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError):
        return {}


def _deep_merge(base: dict, patch: dict) -> dict:
    for k, v in patch.items():
        if isinstance(v, dict) and isinstance(base.get(k), dict):
            _deep_merge(base[k], v)
        else:
            base[k] = v
    return base


def save_overrides(merge: dict[str, Any]) -> dict[str, Any]:
    errs = validate_overrides(merge)
    if errs:
        raise ValueError("; ".join(errs))
    OVERRIDES_PATH.parent.mkdir(parents=True, exist_ok=True)
    with portalocker.Lock(str(OVERRIDES_PATH), "a+", timeout=5) as f:
        f.seek(0)
        try:
            current = json.load(f) if f.read() else {}
        except json.JSONDecodeError:
            current = {}
        merged = _deep_merge(current, merge)
        f.seek(0)
        f.truncate()
        json.dump(merged, f, ensure_ascii=False, indent=2)
    apply_overrides(merged)
    return merged


def apply_overrides(overrides: dict[str, Any]) -> None:
    """原地变更 settings 单例（pydantic 可变，消费方持同一引用）。"""
    sw = overrides.get("scoring_weights")
    if sw:
        for k in WEIGHT_KEYS:
            if k in sw:
                setattr(settings.scoring_weights, k, sw[k])
    pf = overrides.get("profit")
    if pf:
        for k, v in pf.items():
            if hasattr(settings.profit, k):
                setattr(settings.profit, k, v)
    cr = overrides.get("crawl")
    if cr and "force_mock" in cr:
        settings.crawl.force_mock = cr["force_mock"]


def current_config() -> dict[str, Any]:
    sw = settings.scoring_weights
    pf = settings.profit
    return {
        "scoring_weights": {k: getattr(sw, k) for k in WEIGHT_KEYS},
        "profit": {
            "profit_threshold": pf.profit_threshold,
            "platform_commission_rate": pf.platform_commission_rate,
            "logistics_rate": pf.logistics_rate,
            "promotion_rate": pf.promotion_rate,
            "refund_loss_rate": pf.refund_loss_rate,
            "tax_rate": pf.tax_rate,
            "packaging_cost": pf.packaging_cost,
        },
        "crawl": {"force_mock": settings.crawl.force_mock},
    }
```

```python
# src/dianshang/server/jobs.py
"""内存任务存储（线程安全）。"""
from __future__ import annotations

import threading
from dataclasses import dataclass, field
from typing import Any


@dataclass
class JobStatus:
    job_id: str
    status: str = "queued"  # queued|running|done|error|cancelled
    current_node: str | None = None
    completed_nodes: list[str] = field(default_factory=list)
    result: dict[str, Any] | None = None
    error: str | None = None
    errors: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    stale: bool = False


class JobStore:
    def __init__(self) -> None:
        self._jobs: dict[str, JobStatus] = {}
        self._lock = threading.Lock()

    def set(self, job_id: str, status: JobStatus) -> None:
        with self._lock:
            self._jobs[job_id] = status

    def get(self, job_id: str) -> JobStatus | None:
        with self._lock:
            return self._jobs.get(job_id)

    def update(self, job_id: str, **fields) -> JobStatus | None:
        with self._lock:
            st = self._jobs.get(job_id)
            if st is None:
                return None
            for k, v in fields.items():
                setattr(st, k, v)
            return st

    def all(self) -> dict[str, JobStatus]:
        with self._lock:
            return dict(self._jobs)


job_store = JobStore()
```

用 `stream_mode="updates"`：每个 chunk 是 `{node_name: state_update}`，既能拿到节点顺序，又可把每份 update 累积 `.update()` 进一个 dict 得到末态。

```python
# src/dianshang/server/runner.py
"""流式跑图：LangGraph stream_mode=updates 驱动 9 节点进度。"""
from __future__ import annotations

import logging
from typing import Any

from dianshang.agent.graph import selection_graph
from dianshang.agent.state import AgentState
from dianshang.memory import long_term as mem
from dianshang.server import jobs
from dianshang.server.config_store import apply_overrides

logger = logging.getLogger(__name__)

NODE_LABELS: dict[str, str] = {
    "task_decompose": "任务拆解",
    "data_fetch": "数据抓取",
    "data_clean": "数据清洗",
    "analyze": "痛点分析",
    "profit_calc": "利润测算",
    "risk_check": "风险排查",
    "score": "综合评分",
    "report": "报告生成",
    "memory_save": "记忆存储",
}

NODE_ORDER = list(NODE_LABELS.keys())


def build_result(state: AgentState) -> dict[str, Any]:
    profit = state.get("profit") or {}
    return {
        "category": state.get("category", ""),
        "platform": state.get("platform", ""),
        "recommendation": state.get("recommendation", ""),
        "recommendation_reason": state.get("recommendation_reason", ""),
        "total_score": state.get("total_score", 0),
        "scores": state.get("scores", {}),
        "profit": profit,
        "pain_points": (state.get("pain_points") or [])[:5],
        "risk_items": state.get("risk_items", []),
        "red_line_hit": bool(state.get("red_line_hit", False)),
        "report_paths": state.get("report_paths", {}),
        "data_provenance": state.get("data_provenance", {}),
        "errors": state.get("errors", []),
        "warnings": state.get("warnings", []),
    }


def run_job(job_id: str, category: str, platform: str = "pdd",
            overrides: dict[str, Any] | None = None) -> None:
    if overrides:
        apply_overrides(overrides)
    jobs.job_store.update(job_id, status="running", current_node=NODE_ORDER[0])
    initial: AgentState = {"category": category, "platform": platform}
    accumulated: dict[str, Any] = dict(initial)
    try:
        for chunk in selection_graph.stream(initial, stream_mode="updates"):
            for node_name, update in chunk.items():
                if node_name not in NODE_LABELS:
                    continue
                if isinstance(update, dict):
                    accumulated.update(update)
                st = jobs.job_store.get(job_id)
                completed = (st.completed_nodes if st else []) + [node_name]
                jobs.job_store.update(
                    job_id,
                    current_node=node_name,
                    completed_nodes=completed,
                )
                mem.append_progress(job_id, node_name)
    except Exception as exc:  # noqa: BLE001
        logger.exception("任务执行失败 job=%s", job_id)
        jobs.job_store.update(job_id, status="error", error=str(exc))
        mem.release_job(job_id)
        return

    result = build_result(accumulated)  # type: ignore[arg-type]
    jobs.job_store.update(
        job_id,
        status="done",
        result=result,
        errors=accumulated.get("errors", []),
        warnings=accumulated.get("warnings", []),
    )
    mem.release_job(job_id)
```

```python
def list_history(limit: int = 50, offset: int = 0) -> list[dict[str, Any]]:

            "SELECT id, category, platform, recommendation, total_score, gross_margin, "
            "red_line_hit, created_at, report_paths FROM selections "
```

```python
# src/dianshang/server/app.py
"""FastAPI 服务：路由 + 静态挂载 + 启动恢复/卡死清理。"""
from __future__ import annotations

import threading
import time
import uuid
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any

from fastapi import FastAPI, HTTPException
from fastapi.responses import FileResponse, JSONResponse
from pydantic import BaseModel, field_validator

from dianshang.config import settings
from dianshang.memory import long_term as mem
from dianshang.server import config_store, jobs, paths, runner

STATIC_DIR = Path(__file__).parent / "static"
PLATFORMS = {"pdd", "taobao", "tmall", "jd"}
STALE_MINUTES = 30


class ResearchRequest(BaseModel):
    category: str
    platform: str = "pdd"
    overrides: dict[str, Any] | None = None

    @field_validator("platform")
    @classmethod
    def _check_platform(cls, v: str) -> str:
        if v not in PLATFORMS:
            raise ValueError(f"platform 必须是 {PLATFORMS} 之一")
        return v


def _err(code: str, message: str, status: int) -> JSONResponse:
    return JSONResponse(status_code=status, content={"error": {"code": code, "message": message}})


def _cleanup_stale() -> None:
    for jid in mem.stale_jobs(STALE_MINUTES):
        mem.release_job(jid)
        st = jobs.job_store.get(jid)
        if st:
            jobs.job_store.update(jid, status="error", error="任务超时被清理")


def _restore_inflight() -> None:
    """启动时恢复未超时的 running 任务为 stale 标记。"""
    cur = mem.get_running_job()
    if not cur:
        return
    if mem.stale_jobs(STALE_MINUTES):
        _cleanup_stale()
        return
    jid = cur["job_id"]
    completed = mem.load_progress(jid)
    jobs.job_store.set(jobs.JobStatus(
        job_id=jid, status="running", stale=True,
        completed_nodes=completed,
        current_node=completed[-1] if completed else None,
        error="服务重启，任务可能已中断，可取消重跑",
    ))


def _stale_loop() -> None:
    while True:
        time.sleep(300)
        try:
            _cleanup_stale()
        except Exception:  # noqa: BLE001
            pass


def create_app() -> FastAPI:
    app = FastAPI(title="选品调研 Agent")
    mem.init_db()
    _restore_inflight()
    threading.Thread(target=_stale_loop, daemon=True).start()

    @app.get("/api/health")
    def health():
        return {"status": "ok"}

    @app.post("/api/research")
    def research(req: ResearchRequest):
        job_id = uuid.uuid4().hex[:12]
        if not mem.claim_job(job_id, req.category, req.platform):
            return _err("job_running", "已有任务运行中，请先完成或取消", 409)
        jobs.job_store.set(jobs.JobStatus(job_id=job_id, status="queued"))
        t = threading.Thread(
            target=runner.run_job,
            args=(job_id, req.category, req.platform, req.overrides),
            daemon=True,
        )
        t.start()
        return {"job_id": job_id}

    @app.get("/api/jobs/{job_id}")
    def get_job(job_id: str):
        st = jobs.job_store.get(job_id)
        if not st:
            return _err("not_found", "任务不存在", 404)
        return {
            "job_id": st.job_id, "status": st.status,
            "current_node": st.current_node, "completed_nodes": st.completed_nodes,
            "result": st.result, "error": st.error,
            "errors": st.errors, "warnings": st.warnings, "stale": st.stale,
        }

    @app.delete("/api/jobs/{job_id}")
    def cancel_job(job_id: str):
        mem.release_job(job_id)
        jobs.job_store.update(job_id, status="cancelled")
        return {"ok": True}

    @app.get("/api/history")
    def history(limit: int = 50, offset: int = 0):
        if limit < 1 or limit > 200:
            return _err("bad_param", "limit 须在 1–200", 400)
        items = mem.list_history(limit=limit, offset=offset)
        # 总数：再查一次未分页计数
        with mem._connect() as conn:
            total = conn.execute("SELECT COUNT(*) FROM selections").fetchone()[0]
        return {"items": items, "total": total, "limit": limit, "offset": offset}

    @app.get("/api/history/{rid}")
    def history_detail(rid: int):
        with mem._connect() as conn:
            conn.row_factory = __import__("sqlite3").Row
            row = conn.execute("SELECT * FROM selections WHERE id=?", (rid,)).fetchone()
        if not row:
            return _err("not_found", "记录不存在", 404)
        import json
        d = dict(row)
        d["report_paths"] = json.loads(d.get("report_paths") or "{}")
        d["state_json"] = json.loads(d.get("state_json") or "{}")
        return d

    @app.get("/api/reports/{filename}")
    def download_report(filename: str):
        safe = paths.sanitize_report_path(filename)
        if safe is None or not safe.exists():
            return _err("bad_path", "非法或不存在 的报告文件", 400)
        return FileResponse(str(safe), filename=safe.name)

    @app.get("/api/config")
    def get_config():
        return config_store.current_config()

    @app.put("/api/config")
    def put_config(payload: dict[str, Any]):
        errs = config_store.validate_overrides(payload)
        if errs:
            return _err("invalid_config", "; ".join(errs), 400)
        try:
            merged = config_store.save_overrides(payload)
        except ValueError as e:
            return _err("invalid_config", str(e), 400)
        return merged

    # 静态前端
    from fastapi.staticfiles import StaticFiles
    app.mount("/static", StaticFiles(directory=str(STATIC_DIR)), name="static")

    @app.get("/")
    def index():
        return FileResponse(str(STATIC_DIR / "index.html"))

    return app


def main():
    import uvicorn
    uvicorn.run(create_app(), host="127.0.0.1", port=8000)
```
